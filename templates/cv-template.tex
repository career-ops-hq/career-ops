% T1 encoding: without it pdflatex's default OT1 has no glyph for "_" and
% "~", so a CV carrying "snake_case" or "~12" loses those characters in the
% PDF text layer (ATS parsers read that layer, not the screen).
% lmodern is not optional here. fontenc alone switches to the EC fonts,
% whose Type 1 files are not installed everywhere; pdflatex then falls back
% to Type 3 bitmaps with no ToUnicode map, which breaks text extraction for
% the whole document. Latin Modern ships matching Type 1 fonts, so every
% glyph keeps an encoding.

%%%%%%%%%%%%%%%%%%%%%%%%%%%%  Commands  %%%%%%%%%%%%%%%%%%%%%%%%%%%%
\newcommand{\resumeItem}[1]{
  \item\small{
    {#1 \vspace{-3pt}}
  }
}
\newcommand{\resumeSubheading}[4]{
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textbf{#1} & \textbf{\small #2} \\
      \textit{\small#3} & \textit{\small #4} \\
    \end{tabular*}\vspace{-7pt}
}
\newcommand{\resumeSubheadingContinue}[2]{
  \vspace{-3pt}
    \begin{tabular*}{1.0\textwidth}[t]{l@{\extracolsep{\fill}}r}
      \textit{\small#1} & \textit{\small #2} \\
    \end{tabular*}\vspace{-7pt}
}
% The left cell is a paragraph column, not `l`, so an over-long project name
% wraps to a second line instead of running off the right edge of the page.
% `l` cannot break a line at all: a name plus its tech list would push the
% whole row past the margin, with nothing in the build reporting a problem.
% 0.76\textwidth leaves room for the right-hand date before the two cells meet.
\newcommand{\resumeProjectHeading}[2]{
  \vspace{-3pt}\item
    \begin{tabular*}{1.0\textwidth}[t]{@{}>{\raggedright\arraybackslash}p{0.72\textwidth}@{\extracolsep{\fill}}>{\raggedleft\arraybackslash}p{0.24\textwidth}@{}}
      \textbf{#1} & \textbf{\small\raggedleft #2} \\
    \end{tabular*}\vspace{-7pt}
}
% Awards are single-line rows, so they get their own list rather than
% \resumeSubHeadingListStart. Two reasons, both about the gap around them:
% \resumeProjectHeading ends in \vspace{-7pt} to pull the following bullet
% tight under a project title, and an award has no bullets, so that tail drags
% the next award (or the next section title) into the award line; and the
% default itemize itemsep/parsep are what that -7pt was cancelling, so an
% award list left on the defaults opens a paragraph-sized gap between two
% one-line awards. So: a tight list, no negative tail, and a -3pt lead above
% the first award to match what \resumeSubheading does under a section title.
% Every gap then lands inside the 6.6-8.2pt band the other sections use.
\newcommand{\resumeAwardHeading}[2]{
  \item
    \begin{tabular*}{1.0\textwidth}[t]{@{}>{\raggedright\arraybackslash}p{0.72\textwidth}@{\extracolsep{\fill}}>{\raggedleft\arraybackslash}p{0.24\textwidth}@{}}
      \textbf{#1} & \textbf{\small\raggedleft #2} \\
    \end{tabular*}
}
\newcommand{\resumeAwardListStart}{\vspace{-3pt}\begin{itemize}[leftmargin=0.0in, label={}, itemsep=1pt, parsep=0pt, topsep=0pt, partopsep=0pt]}
\newcommand{\resumeAwardListEnd}{\end{itemize}}
\newcommand{\resumeSubItem}[1]{\resumeItem{#1}\vspace{0pt}}
\renewcommand\labelitemi{$\vcenter{\hbox{\tiny$\bullet$}}$}
\renewcommand\labelitemii{$\vcenter{\hbox{\tiny$\bullet$}}$}
\newcommand{\resumeSubHeadingListStart}{\begin{itemize}[leftmargin=0.0in, label={}]}
\newcommand{\resumeSubHeadingListEnd}{\end{itemize}}
\newcommand{\resumeItemListStart}{\begin{itemize}}
\newcommand{\resumeItemListEnd}{\end{itemize}\vspace{0pt}}
